static class Program
{
    [STAThread]
    static void Main(string[] args)
    {
        Application.EnableVisualStyles();
        Application.SetCompatibleTextRenderingDefault(false);

        // 兜底: 任何界面线程上的未处理异常都记进 crash.log 并弹提示, 而不是直接崩掉
        Application.SetUnhandledExceptionMode(UnhandledExceptionMode.CatchException);
        Application.ThreadException += delegate(object s, System.Threading.ThreadExceptionEventArgs e)
        {
            LogCrash(e.Exception);
            MessageBox.Show("程序遇到一个错误, 详情已写入 crash.log:\n\n" + e.Exception.Message,
                "fan_studio", MessageBoxButtons.OK, MessageBoxIcon.Warning);
        };
        AppDomain.CurrentDomain.UnhandledException += delegate(object s, UnhandledExceptionEventArgs e)
        {
            LogCrash(e.ExceptionObject as Exception);
        };

        // 无界面自检: 写日志, 方便在没有桌面的情况下验证这个编译产物
        if (args.Length > 0 && args[0] == "--selftest")
        {
            RunSelfTest();
            return;
        }
        if (args.Length > 0 && args[0] == "--uitest")
        {
            var form = new MainForm(null);
            form.Shown += delegate { form.RunUiTest(); };
            Application.Run(form);
            return;
        }
        try
        {
            Application.Run(new MainForm(args.Length > 0 ? args[0] : null));
        }
        catch (Exception ex)
        {
            LogCrash(ex);
            MessageBox.Show("启动失败:\n\n" + ex, "fan_studio", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }

    static void LogCrash(Exception ex)
    {
        try
        {
            string path = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "crash.log");
            var sb = new StringBuilder();
            sb.AppendLine("=== " + DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss") + " ===");
            sb.AppendLine(ex == null ? "(无异常对象)" : ex.ToString());
            sb.AppendLine();
            File.AppendAllText(path, sb.ToString(), Encoding.UTF8);
        }
        catch { }
    }

    static void RunSelfTest()
    {
        string dir = AppDomain.CurrentDomain.BaseDirectory;
        var log = new StringBuilder();
        try
        {
            var cat = LoadCatalog(dir);
            log.AppendLine("电机库: " + cat.Count);
            string score = File.ReadAllText(Path.Combine(dir, "selftest.fscore"), Encoding.UTF8);
            var sc = ScoreParser.Parse(score, cat, true);
            Planner.Build(sc);
            int fs = 48000;
            var mix = Synth.Render(sc, fs, 0.85, 0.14);
            string wav = Path.Combine(dir, "gui_selftest.wav");
            WavWriter.Write(wav, mix, fs);
            double worst; int det, tot;
            string rep = PitchCheck.Run(sc, mix, fs, out worst, out det, out tot);
            log.AppendLine(string.Format(CultureInfo.InvariantCulture,
                "声部 {0}, 时长 {1:F2}s, 检出 {2}/{3}, 最差 {4:F1} 音分",
                sc.Voices.Count, mix[0].Length / (double)fs, det, tot, worst));
            log.Append(rep);
        }
        catch (Exception ex) { log.AppendLine("失败: " + ex); }
        File.WriteAllText(Path.Combine(dir, "gui_selftest.log"), log.ToString(), Encoding.UTF8);
    }

    public static List<Motor> LoadCatalog(string dir)
    {
        string p = Path.Combine(dir, "motors.txt");
        if (!File.Exists(p)) p = Path.Combine(dir, "..", "motors.txt");
        var list = MotorCatalog.Load(p);
        if (list.Count == 0) list.Add(MotorCatalog.Fallback());
        return list;
    }
}

    // ───────────────────────── 数据 ─────────────────────────
    void FillList()
    {
        _list.BeginUpdate();
        _list.Items.Clear();
        foreach (var m in _catalog)
        {
            if (m.Kind != "revolve") continue;
            var it = new ListViewItem(m.Model);
            it.SubItems.Add(m.Family);
            it.SubItems.Add(m.Blades.ToString(CI));
            it.SubItems.Add(m.MinRpm.ToString("0", CI) + "-" + m.MaxRpm.ToString("0", CI));
            it.SubItems.Add(RangeText(m));
            it.Tag = m;
            it.Checked = m.Enabled;
            it.ToolTipText = m.Note;
            _list.Items.Add(it);
        }
        _list.EndUpdate();
    }

    static string RangeText(Motor m)
    {
        if (m.Kind != "revolve") return "—";
        return HzName(m.LoHz) + "-" + HzName(m.HiHz);
    }

    static string HzName(double hz)
    {
        if (hz <= 0) return "—";
        double midi = 69 + 12 * Math.Log(hz / 440.0, 2);
        return ScoreParser.MidiToName((int)Math.Round(midi));
    }

    void PickFromList()
    {
        if (!_uiReady || _list == null || _nBlades == null) return;
        if (_list.SelectedItems.Count == 0) { _sel = null; return; }
        var m = _list.SelectedItems[0].Tag as Motor;
        if (m == null) return;
        _sel = m;
        // 填值期间必须挡住 ValueChanged 回写, 否则每个控件赋值都会把同一个电机的参数覆盖成半成品
        _filling = true;
        try
        {
            _nBlades.Value = Clamp(m.Blades, _nBlades);
            _nMin.Value = Clamp(m.MinRpm, _nMin);
            _nMax.Value = Clamp(m.MaxRpm, _nMax);
            _nStart.Value = Clamp(m.StartDuty, _nStart);
            _nGlide.Value = Clamp(m.Glide * 1000, _nGlide);
            _nHarm.Value = Clamp(m.Harmonic < 1 ? 1 : m.Harmonic, _nHarm);
            _nTimbre.Value = Clamp(m.Timbre, _nTimbre);
            _nNoiseC.Value = Clamp(m.NoiseCenter, _nNoiseC);
            _nNoiseG.Value = Clamp(m.NoiseGain, _nNoiseG);
            _nPan.Value = Clamp(m.Pan, _nPan);
            _nGain.Value = Clamp(m.Gain, _nGain);
        }
        finally { _filling = false; }
        UpdateRange();
    }

    static decimal Clamp(double v, NumericUpDown n)
    {
        decimal d = (decimal)v;
        if (d < n.Minimum) d = n.Minimum;
        if (d > n.Maximum) d = n.Maximum;
        return d;
    }

    void ApplyParams()
    {
        if (_filling || !_uiReady || _sel == null) return;
        _sel.Blades = (int)_nBlades.Value;
        _sel.MinRpm = (double)_nMin.Value;
        _sel.MaxRpm = (double)_nMax.Value;
        _sel.StartDuty = (double)_nStart.Value;
        _sel.Glide = (double)_nGlide.Value / 1000.0;
        _sel.Harmonic = (double)_nHarm.Value;
        _sel.Timbre = (double)_nTimbre.Value;
        _sel.NoiseCenter = (double)_nNoiseC.Value;
        _sel.NoiseGain = (double)_nNoiseG.Value;
        _sel.Pan = (double)_nPan.Value;
        _sel.Gain = (double)_nGain.Value;
        _sel.LayoutFromFile = true;
        UpdateRange();
        foreach (ListViewItem it in _list.SelectedItems) { it.SubItems[2].Text = _sel.Blades.ToString(CI); it.SubItems[4].Text = RangeText(_sel); }
    }

    void UpdateRange()
    {
        if (_sel == null) { _range.Text = ""; return; }
        string warn = _sel.MaxRpm <= _sel.MinRpm ? "  ⚠ 最高转速必须大于最低" : "";
        _range.Text = string.Format(CI, "能唱 {0}-{1}  ({2:0}-{3:0} Hz){4}",
            HzName(_sel.LoHz), HzName(_sel.HiHz), _sel.LoHz, _sel.HiHz, warn);
    }

    void SyncChecked()
    {
        if (!_uiReady || _list == null) return;
        foreach (ListViewItem it in _list.Items)
        {
            var m = it.Tag as Motor;          // FillList 期间可能拿到还没挂 Tag 的条目
            if (m == null) continue;
            m.Enabled = it.Checked;
        }
        UpdateHint();
    }

    void SetAll(bool on) { foreach (ListViewItem it in _list.Items) it.Checked = on; }
    void FansOnly()
    {
        foreach (ListViewItem it in _list.Items)
        {
            var m = (Motor)it.Tag;
            it.Checked = m.Family == "轴流风机" || m.Family == "离心风机";
        }
    }

    void UpdateHint()
    {
        try
        {
            if (_hint == null || _list == null) return;
            int n = 0;
            foreach (ListViewItem it in _list.Items) if (it.Checked) n++;
            _hint.Text = string.Format("已选 {0} 台" + (n > 8 ? "  (超过 8 台低频会糊, 物理限制)" : ""), n);
        }
        catch { }   // 提示文字是装饰, 绝不该弄崩程序
    }

    // ───────────────────────── 乐谱 ─────────────────────────
    string Template()
    {
        var sb = new StringBuilder();
        sb.AppendLine("# fan_studio 模板 —— 改音符就行, 参数在左边面板调");
        sb.AppendLine("bpm = 120");
        sb.AppendLine();
        int n = 0;
        foreach (ListViewItem it in _list.Items)
        {
            if (!it.Checked) continue;
            var m = (Motor)it.Tag;
            if (n >= 6) break;
            sb.AppendLine("[voice " + m.Key + "]");
            int lo = (int)Math.Round(69 + 12 * Math.Log(m.LoHz / 440.0, 2));
            int hi = (int)Math.Round(69 + 12 * Math.Log(m.HiHz / 440.0, 2));
            int mid = (lo + hi) / 2;
            int[] seq = { 0, 4, 7, 12 };
            var line = new StringBuilder();
            for (int i = 0; i < seq.Length; i++) line.Append(ScoreParser.MidiToName(mid + seq[i]) + ":1 ");
            sb.AppendLine(line.ToString().TrimEnd());
            sb.AppendLine(line.Replace(":1", ":0.5").ToString().TrimEnd());
            sb.AppendLine();
            n++;
        }
        if (n == 0) sb.AppendLine("# 左边先勾几台电机, 再点“插入模板”");
        return sb.ToString();
    }

    void SaveScore()
    {
        var d = new SaveFileDialog();
        d.Filter = "风扇谱 (*.fscore)|*.fscore";
        d.FileName = _scorePath != null ? Path.GetFileName(_scorePath) : "my_fan_song.fscore";
        if (d.ShowDialog() != DialogResult.OK) return;
        try
        {
            File.WriteAllText(d.FileName, _score.Text, new UTF8Encoding(true));
            _scorePath = d.FileName;
            SetStatus("已保存 " + Path.GetFileName(d.FileName));
        }
        catch (Exception ex) { MessageBox.Show("保存失败: " + ex.Message); }
    }

    /// <summary>把当前选中电机的参数写进谱子对应声部(不覆盖音符)。</summary>
    void InsertVoice()
    {
        if (_sel == null) return;
        string header = "[voice " + _sel.Key + "]";
        var lines = new List<string>(_score.Text.Replace("\r\n", "\n").Split('\n'));
        int at = lines.FindIndex(delegate(string s) { return s.Trim() == header; });
        var block = new List<string>();
        block.Add(header);
        block.Add("blades = " + _sel.Blades.ToString(CI));
        block.Add("harmonic = " + ((int)_sel.Harmonic).ToString(CI));
        block.Add("glide = " + _sel.Glide.ToString("0.###", CI));
        block.Add("min_rpm = " + _sel.MinRpm.ToString("0", CI));
        block.Add("max_rpm = " + _sel.MaxRpm.ToString("0", CI));
        block.Add("start_duty = " + _sel.StartDuty.ToString("0.###", CI));
        block.Add("pan = " + _sel.Pan.ToString("0.##", CI));
        block.Add("gain = " + _sel.Gain.ToString("0.##", CI));
        block.Add("noise_center = " + _sel.NoiseCenter.ToString("0", CI));
        block.Add("timbre = " + _sel.Timbre.ToString("0.###", CI));
        block.Add("noise_gain = " + _sel.NoiseGain.ToString("0.###", CI));

        if (at >= 0)
        {
            int end = at + 1;
            while (end < lines.Count && !lines[end].TrimStart().StartsWith("[") && !lines[end].Contains(":"))
            {
                string t = lines[end].Trim();
                bool isParam = t.Length > 0 && t.IndexOf('=') > 0;
                if (!isParam) break;
                end++;
            }
            lines.RemoveRange(at, end - at);
        }
        else { at = lines.Count; }
        lines.InsertRange(at, block);
        _score.Text = string.Join("\r\n", lines.ToArray());
        SetStatus("已把 " + _sel.Key + " 的参数写进谱子");
    }

    // ───────────────────────── 渲染 / 试听 ─────────────────────────
    void Audition()
    {
        if (_sel == null) { SetStatus("先选一台电机"); return; }
        int lo = (int)Math.Round(69 + 12 * Math.Log(_sel.LoHz / 440.0, 2));
        int hi = (int)Math.Round(69 + 12 * Math.Log(_sel.HiHz / 440.0, 2));
        int mid = (lo + hi) / 2;
        var sc = new Score();
        sc.Bpm = 96;
        var v = new Voice();
        v.Motor = _sel;
        v.Label = _sel.Key;
        int[] seq = { 0, 4, 7, 12, 7, 4, 0, 4 };
        for (int i = 0; i < seq.Length; i++)
        {
            var e = new NoteEv();
            int m = mid + seq[i];
            if (m > hi) m = hi;
            if (m < lo) m = lo;
            e.Midi = m; e.Beats = 1;
            v.Raw.Add(e);
        }
        sc.Voices.Add(v);
        PlayScore(sc, _sel.Model);
    }

    void RenderAndPlay()
    {
        if (_busy) return;
        _busy = true;
        _btnRender.Enabled = false;
        SetStatus("渲染中…");
        _bw.RunWorkerAsync(_score.Text);
    }

    string DoRender(string text)
    {
        var sc = ScoreParser.Parse(text, _catalog, true);
        if (sc.Voices.Count == 0)
            return "ERR|谱子里没有任何可识别的 [voice ...] 声部。\n\n可用声部名(点“插入模板”自动生成): " + KeyList();
        Planner.Build(sc);
        int fs = 48000;
        var mix = Synth.Render(sc, fs, 0.85, 0.14);
        double worst; int det, tot;
        string rep = PitchCheck.Run(sc, mix, fs, out worst, out det, out tot);
        string wav = Path.Combine(Path.GetTempPath(), "fan_studio_preview.wav");
        WavWriter.Write(wav, mix, fs);
        var sb = new StringBuilder();
        sb.Append("OK|").Append(wav).Append('|');
        sb.AppendFormat(CI, "{0:F1}s, {1} 声部, 检出 {2}/{3}, 最差 {4:+0.0;-0.0} 音分",
            mix[0].Length / (double)fs, sc.Voices.Count, det, tot, worst);
        sb.Append('|').Append(rep);
        if (sc.ParseWarnings.Count > 0) sb.Append("\n注意: ").Append(sc.ParseWarnings[0]);
        else if (sc.Warnings.Count > 0) sb.Append("\n注意: ").Append(sc.Warnings[0]);
        return sb.ToString();
    }

    string KeyList()
    {
        var sb = new StringBuilder();
        int n = 0;
        foreach (var m in _catalog) { if (m.Kind != "revolve") continue; if (n++ > 12) { sb.Append("…"); break; } sb.Append(m.Key).Append(' '); }
        return sb.ToString();
    }

    void FinishRender(RunWorkerCompletedEventArgs e)
    {
        _busy = false;
        _btnRender.Enabled = true;
        if (e.Error != null) { SetStatus("渲染失败: " + e.Error.Message); MessageBox.Show(e.Error.ToString()); return; }
        string res = (string)e.Result;
        if (res.StartsWith("ERR|")) { SetStatus("渲染失败"); MessageBox.Show(res.Substring(4)); return; }
        string[] parts = res.Split('|');
        SetStatus(parts[2]);
        Play(parts[1]);
        if (parts.Length > 3 && parts[3].Trim().Length > 0)
            Console.WriteLine(parts[3]);
    }

    void PlayScore(Score sc, string title)
    {
        try
        {
            Planner.Build(sc);
            var mix = Synth.Render(sc, 48000, 0.85, 0.14);
            string wav = Path.Combine(Path.GetTempPath(), "fan_studio_audition.wav");
            WavWriter.Write(wav, mix, 48000);
            SetStatus("试听: " + title);
            Play(wav);
        }
        catch (Exception ex) { MessageBox.Show(ex.Message); }
    }

    void Play(string wav)
    {
        try
        {
            _player.Stop();
            _player.SoundLocation = wav;
            _player.Load();
            _player.Play();
        }
        catch (Exception ex) { SetStatus("播放失败: " + ex.Message); }
    }

    void SetStatus(string text)
    {
        var ssl = _status.Tag as ToolStripStatusLabel;
        if (ssl != null) ssl.Text = text; else _status.Text = text;
    }

    /// <summary>无头 UI 冒烟测试: 自动走一遍会崩的交互路径, 结果写 ui_test.log。</summary>
    public void RunUiTest()
    {
        var sb = new StringBuilder();
        string logPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "ui_test.log");
        int step = 0;
        var t = new System.Windows.Forms.Timer();
        t.Interval = 300;
        t.Tick += delegate
        {
            step++;
            try
            {
                if (step == 1)
                {
                    sb.AppendLine("1 列表条目数 = " + _list.Items.Count);
                    _list.Items[0].Selected = true;
                    sb.AppendLine("  选中第0项 -> _sel = " + (_sel == null ? "null" : _sel.Key));
                }
                else if (step == 2)
                {
                    _list.Items[0].Checked = !_list.Items[0].Checked;
                    sb.AppendLine("2 翻转勾选 -> Enabled = " + ((Motor)_list.Items[0].Tag).Enabled);
                }
                else if (step == 3)
                {
                    _list.Items[4].Selected = true;
                    sb.AppendLine("3 切到第4项 -> _sel = " + (_sel == null ? "null" : _sel.Key)
                        + "  音域 = " + _range.Text);
                }
                else if (step == 4)
                {
                    _nGlide.Value = 300;
                    _nTimbre.Value = 1.6M;
                    sb.AppendLine("4 改参数 -> τ = " + _sel.Glide.ToString("0.###", CI)
                        + "  timbre = " + _sel.Timbre.ToString("0.###", CI));
                    InsertVoice();
                    sb.AppendLine("  写入谱子成功, 行数 = " + _score.Lines.Length);
                }
                else if (step == 5)
                {
                    _score.Text = Template();
                    sb.AppendLine("5 生成模板 -> 行数 = " + _score.Lines.Length);
                }
                else if (step == 6)
                {
                    Audition();
                    sb.AppendLine("6 试听已调用");
                }
                else if (step == 7)
                {
                    RenderAndPlay();
                    sb.AppendLine("7 已发起渲染");
                }
                else if (step == 13)
                {
                    sb.AppendLine("8 渲染结束 -> 状态栏: " + ((ToolStripStatusLabel)_status.Tag).Text);
                    sb.AppendLine("RESULT: OK 全部交互路径无异常");
                    t.Stop();
                    File.WriteAllText(logPath, sb.ToString(), Encoding.UTF8);
                    Close();
                }
            }
            catch (Exception ex)
            {
                sb.AppendLine("RESULT: 第 " + step + " 步抛异常");
                sb.AppendLine(ex.ToString());
                t.Stop();
                File.WriteAllText(logPath, sb.ToString(), Encoding.UTF8);
                Close();
            }
        };
        t.Start();
    }
